===== frontend/src/shared/api.ts =====
import { z } from 'zod';

const BASE = import.meta.env.VITE_API_BASE_URL || '';

export const ScanResponseSchema = z.object({
  scan_id: z.string(),
  status: z.string(),
  reported_financial_loss: z.boolean(),
  classification: z.string().optional(),
  confidence: z.string().optional(),
  rule_ids: z.array(z.string()).optional(),
  technical_evidence: z.record(z.any()).optional(),
  social_engineering_evidence: z.record(z.any()).optional(),
  disclaimer: z.string().optional(),
});

export type ScanResponse = z.infer<typeof ScanResponseSchema>;

export async function createScan(content: string, lostMoney: boolean): Promise<ScanResponse> {
  const res = await fetch(`${BASE}/api/v1/scans`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Idempotency-Key': crypto.randomUUID(),
    },
    body: JSON.stringify({
      raw_content: content,
      reported_financial_loss: lostMoney,
      channel: 'web',
    }),
  });
  if (!res.ok) throw new Error('Failed to analyze scan');
  return ScanResponseSchema.parse(await res.json());
}

export async function getScan(scanId: string): Promise<ScanResponse> {
  const res = await fetch(`${BASE}/api/v1/scans/${scanId}`);
  if (!res.ok) throw new Error('Scan not found');
  return ScanResponseSchema.parse(await res.json());
}

export async function deleteScan(scanId: string): Promise<void> {
  const res = await fetch(`${BASE}/api/v1/scans/${scanId}`, { method: 'DELETE' });
  if (!res.ok && res.status !== 404) throw new Error('Deletion failed');
}

export async function getReportUrl(scanId: string): Promise<string> {
  const res = await fetch(`${BASE}/api/v1/scans/${scanId}/report.pdf`);
  if (!res.ok) throw new Error('Failed to get report token');
  const data = await res.json();
  return `${BASE}${data.url}`;
}

export function defang(url: string): string {
  return url.replace(/http/g, 'hxxp').replace(/\./g, '[.]');
}

===== frontend/src/shared/DefangedIndicators.tsx =====
import { useState } from 'react';
import { useTranslation } from 'react-i18next';
import { defang } from './api';

export default function DefangedIndicators({ items }: { items: string[] }) {
  const { t } = useTranslation();
  const [copied, setCopied] = useState(false);

  if (!items.length) return null;

  const defangedItems = items.map((i) => defang(i));

  const copyAll = async () => {
    await navigator.clipboard.writeText(defangedItems.join('\n'));
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className="mt-4 bg-slate-100 p-4 rounded-xl border border-slate-200">
      <div className="flex justify-between items-center mb-2">
        <h4 className="font-semibold text-slate-800 text-sm">{t('result.indicators')}</h4>
        <button
          onClick={copyAll}
          className="text-xs bg-white border border-slate-300 px-2 py-1 rounded hover:bg-slate-50 text-slate-700 font-medium"
        >
          {copied ? 'Copied!' : 'Copy Defanged'}
        </button>
      </div>
      <div className="space-y-1 font-mono text-xs">
        {defangedItems.map((val, idx) => (
          <div key={idx} className="bg-white p-2 rounded border border-slate-200 break-all text-slate-700">
            {val}
          </div>
        ))}
      </div>
    </div>
  );
}

===== frontend/src/shared/history.ts =====
export interface HistoryEntry {
  scan_id: string;
  status: string;
  classification?: string;
  date: string;
}

const STORAGE_KEY = 'cybersathi_history';

export function readHistory(): HistoryEntry[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

export function saveHistory(entry: HistoryEntry): void {
  try {
    const list = readHistory().filter((i) => i.scan_id !== entry.scan_id);
    localStorage.setItem(STORAGE_KEY, JSON.stringify([entry, ...list].slice(0, 50)));
  } catch {
    // LocalStorage quota handling
  }
}

export function removeHistory(scanId: string): void {
  try {
    const list = readHistory().filter((i) => i.scan_id !== scanId);
    localStorage.setItem(STORAGE_KEY, JSON.stringify(list));
  } catch {
    // LocalStorage quota handling
  }
}

import { useTranslation } from 'react-i18next';
import { Link, useParams } from 'react-router-dom';
import { useQuery } from '@tanstack/react-query';
import { getReportUrl, getScan } from '../../shared/api';
import DefangedIndicators from '../../shared/DefangedIndicators';
import HelplineBanner from '../../shared/HelplineBanner';

const BADGE_COLORS: Record<string, string> = {
  CONFIRMED_MALICIOUS: 'bg-red-100 text-red-800 border-red-300',
  HIGH_RISK: 'bg-orange-100 text-orange-800 border-orange-300',
  SUSPICIOUS: 'bg-amber-100 text-amber-800 border-amber-300',
  NO_KNOWN_THREAT_DETECTED: 'bg-emerald-100 text-emerald-800 border-emerald-300',
  INCONCLUSIVE: 'bg-slate-100 text-slate-800 border-slate-300',
};

export default function ResultPage() {
  const { t } = useTranslation();
  const { scanId } = useParams();

  const { data: scan, isLoading, error } = useQuery({
    queryKey: ['scan', scanId],
    queryFn: () => getScan(scanId!),
    enabled: Boolean(scanId),
  });

  const handleDownload = async () => {
    try {
      const url = await getReportUrl(scanId!);
      window.open(url, '_blank');
    } catch {
      alert('Could not download report.');
    }
  };

  if (isLoading) return <div className="text-center py-20">Loading analysis...</div>;
  if (error || !scan) return <div className="text-center py-20 text-red-600">Scan not found.</div>;

  const classification = scan.classification || 'INCONCLUSIVE';
  const badgeStyle = BADGE_COLORS[classification] || BADGE_COLORS.INCONCLUSIVE;

  // Extract raw indicators for defanged rendering
  const extractedDomains: string[] = [];
  const techItems = scan.technical_evidence?.items || [];
  techItems.forEach((item: any) => {
    if (Array.isArray(item.domains)) extractedDomains.push(...item.domains);
  });

  return (
    <div className="max-w-3xl mx-auto px-4 py-8">
      {scan.reported_financial_loss && <HelplineBanner />}

      <div className="bg-white p-6 sm:p-8 rounded-2xl shadow-sm border border-slate-200 space-y-6">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 border-b pb-6">
          <div>
            <span className={`inline-block px-3 py-1 rounded-full text-xs font-bold border ${badgeStyle} mb-2`}>
              {classification}
            </span>
            <h1 className="text-2xl font-extrabold text-slate-900">{t(`assessment.${classification.toLowerCase()}`)}</h1>
          </div>
          <button
            onClick={handleDownload}
            className="bg-brand-600 hover:bg-brand-700 text-white px-5 py-2.5 rounded-xl font-medium text-sm transition shadow flex items-center gap-2 justify-center"
          >
            {t('result.download')}
          </button>
        </div>

        <div>
          <h3 className="font-semibold text-slate-800 mb-1">{t('result.confidence')}</h3>
          <p className="text-sm text-slate-600">{scan.confidence || 'LOW'}</p>
        </div>

        {scan.rule_ids && scan.rule_ids.length > 0 && (
          <div>
            <h3 className="font-semibold text-slate-800 mb-2">{t('result.rules')}</h3>
            <div className="flex flex-wrap gap-2">
              {scan.rule_ids.map((r) => (
                <span key={r} className="bg-slate-100 text-slate-700 text-xs font-mono px-2.5 py-1 rounded border border-slate-200">
                  {r}
                </span>
              ))}
            </div>
          </div>
        )}

        <DefangedIndicators items={extractedDomains} />

        <div className="p-4 bg-slate-50 border border-slate-200 rounded-xl text-xs text-slate-500 space-y-1">
          <p className="font-semibold text-slate-700">Notice:</p>

}

import { useState } from 'react';
import { useTranslation } from 'react-i18next';
import { Link } from 'react-router-dom';
import { deleteScan } from '../../shared/api';
import { readHistory, removeHistory } from '../../shared/history';

export default function HistoryPage() {
  const { t } = useTranslation();
  const [history, setHistory] = useState(readHistory);

  const handleDelete = async (scanId: string) => {
    try {
      await deleteScan(scanId);
    } catch {
      // Local deletion fallback
    }
    removeHistory(scanId);
    setHistory(readHistory());
  };

  return (
    <div className="max-w-2xl mx-auto px-4 py-8">
      <h2 className="text-2xl font-bold text-slate-900 mb-6">{t('history.title')}</h2>
      {!history.length ? (
        <div className="bg-white p-8 rounded-xl border border-slate-200 text-center text-slate-500">
          {t('history.empty')}
        </div>
      ) : (
        <div className="space-y-3">
          {history.map((item) => (

from __future__ import annotations

from pathlib import Path
import yaml

_RULES_PATH = Path(__file__).parent / "rules.yaml"
_rules_cache: dict | None = None

SUSPICIOUS_TLDS = {".top", ".xyz", ".click", ".info", ".buzz", ".live", ".guru", ".link"}


def load_rules() -> dict:
    global _rules_cache
    if _rules_cache is None:
        with _RULES_PATH.open(encoding="utf-8") as fh:
            _rules_cache = yaml.safe_load(fh)
    return _rules_cache


def evaluate_social_engineering(text: str) -> tuple[list[dict], list[str]]:
    ruleset = load_rules()
    evidence: list[dict] = []
    matched: list[str] = []
    lowered = text.lower()

    for rule in ruleset.get("rules", []):
        if rule["category"] != "social_engineering":
            continue
        hits = [p for p in rule.get("patterns", []) if p.lower() in lowered]
        if hits:
            evidence.append({"rule_id": rule["id"], "description": rule["description"], "hits": hits})
            matched.append(rule["id"])
    return evidence, matched


def evaluate_technical(
    domains: list[str],
    observations: list[dict] | None = None,
    domain_age_days: int | None = None,
) -> tuple[list[dict], list[str]]:
    observations = observations or []
    evidence: list[dict] = []
    matched: list[str] = []

    for obs in observations:
        if obs.get("outcome") == "MATCH":
            evidence.append({"provider": obs.get("provider"), "outcome": "MATCH"})
    if any(o.get("outcome") == "MATCH" for o in observations):
        matched.append("TECH_PROVIDER_MATCH")

    if domains:
        evidence.append({"domains": domains})
        matched.append("TECH_URL_PRESENT")
        for d in domains:
            d_lower = d.lower()
            if any(d_lower.endswith(tld) for tld in SUSPICIOUS_TLDS):
                evidence.append({"domain": d, "signal": "suspicious_tld"})
                if "TECH_SUSPICIOUS_TLD" not in matched:
                    matched.append("TECH_SUSPICIOUS_TLD")

    if domain_age_days is not None and domain_age_days < 30:
        evidence.append({"domain_age_days": domain_age_days, "signal": "new_domain"})
        matched.append("TECH_NEW_DOMAIN")

    return evidence, matched


def classify(
    *,
    has_verified_match: bool,
    corroborating_categories: set[str],
    se_rule_count: int,
    has_weak_technical: bool,
    all_providers_responded: bool = True,
) -> str:
    """Deterministic classification order (first match wins).

    1. CONFIRMED_MALICIOUS: Match from verified threat intel.
    2. HIGH_RISK: >= 2 independent corroborating categories (e.g. SE + Technical).
    3. SUSPICIOUS: Any SE rule or weak technical signal.
    4. NO_KNOWN_THREAT_DETECTED: Evaluated clean with no matches.
    5. INCONCLUSIVE: Insufficient evidence.
    """
    if has_verified_match:
        return "CONFIRMED_MALICIOUS"
    if len(corroborating_categories) >= 2 or se_rule_count >= 2:
        return "HIGH_RISK"
    if se_rule_count >= 1 or has_weak_technical:
        return "SUSPICIOUS"
    if all_providers_responded:
        return "NO_KNOWN_THREAT_DETECTED"
    return "INCONCLUSIVE"


EXPLANATION_KEYS = {
    "CONFIRMED_MALICIOUS": "assessment.confirmed_malicious",
    "HIGH_RISK": "assessment.high_risk",
    "SUSPICIOUS": "assessment.suspicious",
    "NO_KNOWN_THREAT_DETECTED": "assessment.no_known_threat",
    "INCONCLUSIVE": "assessment.inconclusive",
}

from __future__ import annotations

import logging
from assessment.engine import (
    EXPLANATION_KEYS,
    classify,
    evaluate_social_engineering,
    evaluate_technical,
    load_rules,
)
from assessment.models import ScanAssessment
from indicators.models import IndicatorObservation, ThreatIndicatorStatus, threat_indicators_for

logger = logging.getLogger(__name__)


def build_assessment(submission, observations: list[dict] | None = None) -> dict:
    observations = observations or []
    text = submission.raw_content
    se_evidence, se_rules = evaluate_social_engineering(text)

    domains = [
        ind.normalised_value
        for ind in submission.indicators.all()
        if ind.indicator_type == "DOMAIN"
    ]

    tech_evidence, tech_rules = evaluate_technical(domains, observations)

    has_verified_match = any(
        o.get("outcome") == "MATCH" and o.get("provider") in {"local_intel", "phishtank"}
        for o in observations
    )

    categories: set[str] = set()
    if has_verified_match:
        categories.add("verified_threat_intel")
    elif any(o.get("outcome") == "MATCH" for o in observations):
        categories.add("technical")
    if se_rules:
        categories.add("social_engineering")
    if "TECH_SUSPICIOUS_TLD" in tech_rules or "TECH_NEW_DOMAIN" in tech_rules:
        categories.add("technical_suspicious")

    has_weak = bool(
        "TECH_SUSPICIOUS_TLD" in tech_rules
        or "TECH_NEW_DOMAIN" in tech_rules
        or "TECH_PROVIDER_MATCH" in tech_rules
    )

    classification_result = classify(
        has_verified_match=has_verified_match,
        corroborating_categories=categories,
        se_rule_count=len(se_rules),
        has_weak_technical=has_weak,
        all_providers_responded=True,
    )

    confidence = (
        "HIGH"
        if (has_verified_match or len(categories) >= 2)
        else ("MEDIUM" if (se_rules or has_weak) else "LOW")
    )
    ruleset = load_rules()

    return {
        "classification": classification_result,
        "technical_evidence": {"items": tech_evidence},
        "social_engineering_evidence": {"items": se_evidence},
        "confidence": confidence,
        "rule_ids": sorted(set(se_rules + tech_rules)),
        "ruleset_version": ruleset.get("version", "2.0"),
        "explanation_key": EXPLANATION_KEYS[classification_result],
    }


def store_assessment(submission, observations: list[dict] | None = None) -> ScanAssessment:
    payload = build_assessment(submission, observations)
    assessment, _ = ScanAssessment.objects.update_or_create(
        submission=submission, defaults=payload
    )
    submission.status = "COMPLETED"
    submission.save(update_fields=["status"])
    return assessment


def run_assessment(scan_id: str) -> ScanAssessment:
    from submissions.models import ScamSubmission

    submission = ScamSubmission.objects.get(id=scan_id)
    inds = list(submission.indicators.all())
    pairs = {(i.indicator_type, bytes(i.digest)) for i in inds}
    threat_rows = threat_indicators_for(pairs)

    obs = []
    for row in threat_rows:
        if row.status in (ThreatIndicatorStatus.CORROBORATED, ThreatIndicatorStatus.CONFIRMED):
            obs.append({"provider": "local_intel", "outcome": "MATCH"})
        for o in row.observations.all():
            obs.append({"provider": o.provider, "outcome": o.outcome})

    return store_assessment(submission, obs)

===== backend/core/crypto.py =====
from __future__ import annotations

import base64
import hashlib
import hmac
from cryptography.fernet import Fernet
from django.conf import settings

_fernet: Fernet | None = None


def _get_fernet() -> Fernet:
    global _fernet
    if _fernet is None:
        key = (getattr(settings, "FIELD_ENCRYPTION_KEY", "") or "").strip()
        if not key:
            key = Fernet.generate_key().decode()
        _fernet = Fernet(key.encode())
    return _fernet


def encrypt(plaintext: str) -> str:
    """Encrypt raw sensitive content at rest."""
    if not plaintext:
        return ""
    return _get_fernet().encrypt(plaintext.encode()).decode()


def decrypt(ciphertext: str) -> str:
    """Decrypt raw content. Returns empty string if missing."""
    if not ciphertext:
        return ""
    try:
        return _get_fernet().decrypt(ciphertext.encode()).decode()
    except Exception:
        return ""


def hmac_digest(value: str, key_version: str | None = None) -> tuple[bytes, str]:
    """Calculate HMAC-SHA256 digest to protect low-entropy identifiers."""
    key_b64 = (getattr(settings, "IDENTIFIER_HMAC_KEY", "") or "").strip()
    version = key_version or getattr(settings, "IDENTIFIER_HMAC_KEY_ID", "v1")
    if not key_b64:
        key = b"dev-key-change-immediately"
    else:
        try:
            key = base64.b64decode(key_b64)
        except Exception:
            key = key_b64.encode()
    digest = hmac.new(key, value.encode("utf-8"), hashlib.sha256).digest()
    return digest, version

from __future__ import annotations

import logging
import time

logger = logging.getLogger(__name__)


class SecurityHeadersMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        response = self.get_response(request)
        response["Content-Security-Policy"] = (
            "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; "
            "img-src 'self' data:; connect-src 'self'; frame-ancestors 'none'; base-uri 'self'"
        )
        response["X-Content-Type-Options"] = "nosniff"
        response["X-Frame-Options"] = "DENY"
        response["Referrer-Policy"] = "strict-origin-when-cross-origin"
        return response


class JsonLoggingMiddleware:
    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        start = time.perf_counter()
        response = self.get_response(request)
        elapsed = time.perf_counter() - start

        logger.info(
            "api_request",
            extra={
                "method": request.method,
                "path": request.path,
                "status": response.status_code,
                "latency_ms": round(elapsed * 1000, 2),
            },
        )
        return response

        forwarded = (request.META.get("HTTP_X_FORWARDED_FOR") or "").split(",")[0].strip()
        ident = (
            request.META.get("HTTP_X_REAL_IP")
            or forwarded
            or request.META.get("REMOTE_ADDR", "127.0.0.1")
        )
        return self.cache_format % {"scope": self.scope, "ident": ident}

class ThreatIndicatorStatus(models.TextChoices):
    REPORTED = "REPORTED"
    UNDER_REVIEW = "UNDER_REVIEW"
    CORROBORATED = "CORROBORATED"
    CONFIRMED = "CONFIRMED"
    EXPIRED = "EXPIRED"


_TRANSITIONS: dict[str, list[str]] = {
    ThreatIndicatorStatus.REPORTED: [ThreatIndicatorStatus.UNDER_REVIEW, ThreatIndicatorStatus.EXPIRED],
    ThreatIndicatorStatus.UNDER_REVIEW: [
        ThreatIndicatorStatus.CORROBORATED,
        ThreatIndicatorStatus.EXPIRED,
    ],
    ThreatIndicatorStatus.CORROBORATED: [
        ThreatIndicatorStatus.CONFIRMED,
        ThreatIndicatorStatus.EXPIRED,
    ],
    ThreatIndicatorStatus.CONFIRMED: [ThreatIndicatorStatus.EXPIRED],
    ThreatIndicatorStatus.EXPIRED: [],
}


class ThreatIndicator(models.Model):
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    indicator_type = models.CharField(max_length=16)
    digest = models.BinaryField()
    status = models.CharField(
        max_length=16,
        choices=ThreatIndicatorStatus.choices,
        default=ThreatIndicatorStatus.REPORTED,
    )
    first_seen = models.DateTimeField(auto_now_add=True)
    last_seen = models.DateTimeField(auto_now=True)
    expires_at = models.DateTimeField(null=True, blank=True)

    class Meta:
        constraints = [
            models.UniqueConstraint(fields=["indicator_type", "digest"], name="unique_type_digest"),
        ]
        indexes = [
            models.Index(fields=["indicator_type", "status"]),
        ]

    def transition_to(self, new_status: str, *, trusted: bool = False) -> None:
        allowed = _TRANSITIONS.get(self.status, [])
        if new_status not in allowed:
            raise ValueError(f"Illegal transition {self.status} -> {new_status}")
        if new_status in (ThreatIndicatorStatus.CORROBORATED, ThreatIndicatorStatus.CONFIRMED) and not trusted:
            raise ValueError("Promotion requires trusted source or examiner review")
        self.status = new_status
        self.save(update_fields=["status", "last_seen"])


def threat_indicators_for(
    pairs: Iterable[tuple[str, bytes]],
) -> QuerySet[ThreatIndicator]:
    query = Q()
    for indicator_type, digest in pairs:
        query |= Q(indicator_type=indicator_type, digest=digest)
    if not query:
        return ThreatIndicator.objects.none()
    return ThreatIndicator.objects.filter(query)


def upsert_indicator(indicator_type: str, digest: bytes) -> tuple[ThreatIndicator, bool]:
    with transaction.atomic():
        obj, created = ThreatIndicator.objects.get_or_create(
            indicator_type=indicator_type,
            digest=digest,
            defaults={"status": ThreatIndicatorStatus.REPORTED},
        )
        if not created:
            obj.last_seen = timezone.now()
            obj.save(update_fields=["last_seen"])
        return obj, created

def signed_url(scan_id: str, max_age: int = 900) -> dict:
    signer = TimestampSigner(salt="report")
    token = signer.sign_object({"scan_id": str(scan_id)})
    return {
        "url": f"/api/v1/scans/{scan_id}/report.pdf?token={quote(token)}",
        "expires_in": max_age,
    }


def verify_token(token: str, scan_id: str, max_age: int = 900) -> bool:
    try:
        signer = TimestampSigner(salt="report")
        data = signer.unsign_object(token, max_age=max_age)
        return str(data.get("scan_id")) == str(scan_id)
    except BadSignature:
        return False

@extend_schema(
    parameters=[
        OpenApiParameter(
            name="token",
            type=OpenApiTypes.STR,
            location=OpenApiParameter.QUERY,
            required=False,
            description="Signed token for direct PDF download.",
        )
    ],
    responses={200: OpenApiTypes.OBJECT},
)
@api_view(["GET"])
@throttle_classes([ReportRateThrottle])
def report_pdf(request: Request, scan_id: str) -> Response | HttpResponse:
    try:
        ScamSubmission.objects.get(id=scan_id)
    except (ScamSubmission.DoesNotExist, ValueError):
        return problem("Scan not found", 404)

    token = request.query_params.get("token", "")
    if not token:
        return Response(signed_url(scan_id))

    if not verify_token(token, scan_id):
        return problem("Invalid or expired token", 403)

    pdf_bytes = generate_pdf(scan_id)
    return HttpResponse(pdf_bytes, content_type="application/pdf")

from __future__ import annotations

import re
import unicodedata
from urllib.parse import urlparse

_ZERO_WIDTH = {"\u200b", "\u200c", "\u200d", "\u2060", "\ufeff", "\u00ad"}
_ALLOWED_URL_SCHEMES = {"http", "https"}

_URL_RE = re.compile(r"https?://[^\s\"'<>\]\)}]+|www\.[^\s\"'<>\]\)}]+", re.IGNORECASE)
_EMAIL_RE = re.compile(r"[\w.+-]+@[\w-]+\.[\w.-]+")
_PHONE_RE = re.compile(r"(?:\+91[\s-]?|91[\s-]?|0)?[6-9](?:[\s-]?\d){9}")
_UPI_RE = re.compile(r"[\w.\-]{2,}@[a-zA-Z]{2,}")
_IP_RE = re.compile(r"(?:(?:25[0-5]|2[0-4]\d|1\d\d|[1-9]?\d)\.){3}(?:25[0-5]|2[0-4]\d|1\d\d|[1-9]?\d)")
_DEVANAGARI_RE = re.compile(r"[\u0900-\u097F]")


def normalise_text(raw: str, max_chars: int = 5000) -> str:
    if len(raw) > max_chars:
        raise ValueError(f"Input exceeds maximum of {max_chars} characters")
    text = unicodedata.normalize("NFKC", raw)
    cleaned: list[str] = []
    for ch in text:
        if ch in _ZERO_WIDTH:
            continue
        if unicodedata.category(ch)[0] == "C":
            if ch in "\n\r\t":
                cleaned.append(" ")
            continue
        cleaned.append(ch)
    return re.sub(r"[ \t]+", " ", "".join(cleaned)).strip()


def detect_language(text: str) -> str:
    has_dev = bool(_DEVANAGARI_RE.search(text))
    has_latin = bool(re.search(r"[A-Za-z]", text))
    if has_dev and has_latin:
        return "mixed"
    if has_dev:
        return "hi"
    return "en"


def extract_urls(text: str) -> list[str]:
    urls: list[str] = []
    for m in _URL_RE.finditer(text):
        candidate = m.group(0).rstrip(".,;:!?)]}")
        norm = normalise_url(candidate)
        if norm:
            urls.append(norm)
    return urls


def normalise_url(raw: str) -> str | None:
    candidate = raw.strip()
    if not candidate or any(ch.isspace() for ch in candidate):
        return None
    if "://" not in candidate:
        candidate = "https://" + candidate
    try:
        parsed = urlparse(candidate)
        hostname = parsed.hostname
        port = parsed.port
    except ValueError:
        return None

    scheme = parsed.scheme.lower()
    if scheme not in _ALLOWED_URL_SCHEMES or not hostname:
        return None
    if parsed.username or parsed.password:
        return None  # Embedded credentials rejected

    host = hostname.lower().rstrip(".")
    try:
        host = host.encode("idna").decode("ascii")
    except UnicodeError:
        return None

    if not host:
        return None

    netloc_host = f"[{host}]" if ":" in host else host
    default_port = (scheme == "https" and port == 443) or (scheme == "http" and port == 80)
    netloc = netloc_host if (port is None or default_port) else f"{netloc_host}:{port}"
    path = parsed.path or "/"
    return f"{scheme}://{netloc}{path}"


def extract_domain(url: str) -> str | None:
    try:
        return urlparse(url).hostname
    except Exception:
        return None


def extract_emails(text: str) -> list[str]:
    return [m.group(0).lower() for m in _EMAIL_RE.finditer(text)]


def extract_phones(text: str) -> list[str]:
    phones: list[str] = []
    for m in _PHONE_RE.finditer(text):
        digits = re.sub(r"\D", "", m.group(0))
        if len(digits) == 12 and digits.startswith("91"):
            digits = digits[2:]
        elif len(digits) == 11 and digits.startswith("0"):
            digits = digits[1:]
        if len(digits) == 10:
            phones.append(digits)
    return phones


def extract_upi_ids(text: str) -> list[str]:
    return [
        m.group(0).lower()
        for m in _UPI_RE.finditer(text)
        if not m.group(0).lower().endswith((".com", ".in", ".org", ".net"))
    ]


def extract_ips(text: str) -> list[str]:
    return [m.group(0) for m in _IP_RE.finditer(text)]

from __future__ import annotations

import logging
from django.conf import settings
from django.db import transaction
from assessment.services import run_assessment
from core.crypto import encrypt, hmac_digest
from submissions.models import ExtractedIndicator, ScamSubmission, WorkflowEvent
from submissions import normalisation as norm

logger = logging.getLogger(__name__)


def create_submission(
    raw_content: str, channel: str = "web", reported_financial_loss: bool = False
) -> ScamSubmission:
    limit = getattr(settings, "MAX_INPUT_CHARS", 5000)
    text = norm.normalise_text(raw_content, max_chars=limit)
    content_digest, _ = hmac_digest(text)

    with transaction.atomic():
        submission = ScamSubmission.objects.create(
            channel=channel,
            input_type="url" if norm.extract_urls(text) else "text",
            reported_financial_loss=reported_financial_loss,
        )
        submission.raw_content = text
        submission.content_hash = content_digest
        submission.save()

        indicators: list[tuple[str, str]] = []
        for url in norm.extract_urls(text):
            indicators.append(("URL", url))
            domain = norm.extract_domain(url)
            if domain:
                indicators.append(("DOMAIN", domain))
        for email in norm.extract_emails(text):
            indicators.append(("EMAIL", email))
        for phone in norm.extract_phones(text):
            indicators.append(("PHONE", phone))
        for upi in norm.extract_upi_ids(text):
            indicators.append(("UPI", upi))
        for ip in norm.extract_ips(text):
            indicators.append(("IP", ip))

        seen = set()
        unique_indicators = []
        for pair in indicators:
            if pair not in seen:
                seen.add(pair)
                unique_indicators.append(pair)

        for itype, value in unique_indicators[:50]:
            digest, version = hmac_digest(value)
            ExtractedIndicator.objects.get_or_create(
                submission=submission,
                indicator_type=itype,
                digest=digest,
                defaults={
                    "key_version": version,
                    "normalised_value_encrypted": encrypt(value),
                },
            )

        WorkflowEvent.objects.create(
            scan=submission,
            event_type="submission_created",
            payload_summary={"indicator_count": len(unique_indicators)},
        )

    return submission


def start_scan_pipeline(submission: ScamSubmission):
    """Synchronous in-process evaluation pipeline."""
    return run_assessment(str(submission.id))

def _scan_payload(submission: ScamSubmission) -> dict:
    payload = {
        "scan_id": str(submission.id),
        "status": submission.status,
        "reported_financial_loss": submission.reported_financial_loss,
    }
    assessment = ScanAssessment.objects.filter(submission=submission).first()
    if assessment:
        payload.update(
            {
                "classification": assessment.classification,
                "technical_evidence": assessment.technical_evidence,
                "social_engineering_evidence": assessment.social_engineering_evidence,
                "confidence": assessment.confidence,
                "rule_ids": assessment.rule_ids,
                "ruleset_version": assessment.ruleset_version,
                "explanation_key": assessment.explanation_key,
                "disclaimer": (
                    "NO_KNOWN_THREAT_DETECTED does not guarantee safety or legitimacy. "
                    "Always exercise caution with unsolicited links or requests."
                    if assessment.classification == "NO_KNOWN_THREAT_DETECTED"
                    else ""
                ),
            }
        )
    return payload


@extend_schema(request=ScanCreateSerializer, responses={201: ScanDetailSerializer})
@api_view(["POST"])
@throttle_classes([SubmissionRateThrottle])
def create_scan(request: Request) -> Response:
    serializer = ScanCreateSerializer(data=request.data)
    if not serializer.is_valid():
        return problem("Validation failed", 400, str(serializer.errors))

    data = serializer.validated_data
    idem_key = request.headers.get("Idempotency-Key", "")
    if idem_key:
        cached_id = cache.get(f"idem:{idem_key}")
        if cached_id:
            try:
                existing = ScamSubmission.objects.get(id=cached_id)
                return Response(_scan_payload(existing), status=status.HTTP_200_OK)
            except ScamSubmission.DoesNotExist:
                pass

    limit = getattr(settings, "MAX_INPUT_CHARS", 5000)
    text = norm.normalise_text(data["raw_content"], max_chars=limit)
    digest, _ = hmac_digest(text)

    submission = create_submission(
        raw_content=data["raw_content"],
        channel=data.get("channel", "web"),
        reported_financial_loss=data.get("reported_financial_loss", False),
    )

    if idem_key:
        cache.set(f"idem:{idem_key}", str(submission.id), timeout=900)

    # Execute synchronous scan immediately
    start_scan_pipeline(submission)
    submission.refresh_from_db()

    return Response(_scan_payload(submission), status=status.HTTP_201_CREATED)


@extend_schema(responses={200: ScanDetailSerializer, 204: OpenApiTypes.OBJECT})
@api_view(["GET", "DELETE"])
def scan_detail(request: Request, scan_id: str) -> Response:
    try:
        submission = ScamSubmission.objects.get(id=scan_id)
    except (ScamSubmission.DoesNotExist, ValueError):
        return problem("Scan not found", 404)

    if request.method == "DELETE":
        submission.delete_content()
        submission.delete()
        return Response(status=status.HTTP_204_NO_CONTENT)

    return Response(_scan_payload(submission))


@extend_schema(responses={200: OpenApiTypes.OBJECT})
@api_view(["GET"])
def get_evidence(request: Request, scan_id: str) -> Response:
    try:
        submission = ScamSubmission.objects.get(id=scan_id)
    except (ScamSubmission.DoesNotExist, ValueError):
        return problem("Scan not found", 404)

    indicators = [
        {"type": ind.indicator_type, "normalised_value": ind.normalised_value}
        for ind in submission.indicators.all()
    ]
    assessment = ScanAssessment.objects.filter(submission=submission).first()

    return Response(
        {
            "scan_id": str(submission.id),
            "indicators": indicators,
            "assessment": {
                "classification": assessment.classification if assessment else None,
                "rule_ids": assessment.rule_ids if assessment else [],
            },
        }
    )
